```csharp
using System.Security.Claims;
using System.Text.Encodings.Web;
using Collector.Api.Services;
using Microsoft.AspNetCore.Authentication;
using Microsoft.Extensions.Options;

namespace Collector.Api.Auth;

/// <summary>
/// Authenticates the Discord ingest route. It reads <c>x-api-key</c> and accepts only a live
/// key whose scope is <see cref="DiscordIngestAuth.IngestScope"/>: never the static admin key
/// (it is not an api.db key), never a full key, never a JWT (the Authorization header is not
/// read). The principal carries the owner's id and name and the scope claim, and never the
/// Admin role, even when the owner is an admin.
/// </summary>
public sealed class DiscordIngestKeyAuthHandler : AuthenticationHandler<ApiKeySchemeOptions>
{
    private readonly ApiKeyService _apiKeyService;

    public DiscordIngestKeyAuthHandler(
        IOptionsMonitor<ApiKeySchemeOptions> options,
        ILoggerFactory logger,
        UrlEncoder encoder,
        ApiKeyService apiKeyService)
        : base(options, logger, encoder)
    {
        _apiKeyService = apiKeyService;
    }

    protected override async Task<AuthenticateResult> HandleAuthenticateAsync()
    {
        // Header only, like the ApiKey scheme: a key in a query string ends up in logs.
        if (!Request.Headers.TryGetValue("x-api-key", out var headerValue)
            || string.IsNullOrWhiteSpace(headerValue.ToString()))
            return AuthenticateResult.NoResult();

        var validation = await _apiKeyService.ValidateAsync(headerValue.ToString(), Context.RequestAborted);
        if (validation is null)
            return AuthenticateResult.Fail("Invalid API key");
        if (validation.Scope != DiscordIngestAuth.IngestScope)
            return AuthenticateResult.Fail("Not a discord:ingest key");
        if (validation.User.IsBanned)
            return AuthenticateResult.Fail("Account is banned");

        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, validation.User.Id.ToString()),
            new Claim(ClaimTypes.Name, validation.User.Username),
            new Claim(DiscordIngestAuth.ScopeClaimType, DiscordIngestAuth.IngestScope),
        };
        var principal = new ClaimsPrincipal(new ClaimsIdentity(claims, Scheme.Name));
        return AuthenticateResult.Success(new AuthenticationTicket(principal, Scheme.Name));
    }
}
```
